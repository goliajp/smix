### S1. `InputChannel` protocol + `DigitizerChannel` + `IndigoChannel` + `ChannelDispatcher` + 单测（Swift XCTest 红→绿）

    ```swift
    import Foundation

    /// Stable HID dispatch channel identifier — matches the wire-ABI `path`
    /// field on `simx-host-hid tap` (C3 `indigo9` / C4 `digitizer`).
    public enum InputChannelId: String, Equatable, CaseIterable {
      case digitizer    // C4 IOHIDEvent-tree path
      case indigo9      // C3 9-arg IndigoHIDMessageForMouseNSEvent path
    }

    /// Wraps either `IOHIDDigitizerTap` (digitizer) or `IndigoHIDDispatcher`
    /// (indigo9) behind a single API so the CLI dispatch + probe + future
    /// (C6+) Driver integration go through one site.
    public protocol InputChannel {
      var id: InputChannelId { get }
      var name: String { get }                    // == id.rawValue (kept on the
                                                  //  protocol so non-rawValue
                                                  //  channel names stay open)
      /// Symbols required by this channel. Used by the probe path:
      /// missing any → the channel reports `available: false`.
      var requiredSymbolNames: [String] { get }

      /// Real tap dispatch. Mirrors `IOHIDDigitizerTap.tapNormalized` /
      /// `IndigoHIDDispatcher.tapNormalized` arg shape so C6 Driver work has
      /// one place to call. Channels reuse existing channel-class function
      /// bodies internally — InputChannel impls are wrappers.
      func tap(
        udid: String,
        x: Double, y: Double,
        developerDir: String,
        coreSimulatorHandle: UnsafeMutableRawPointer,
        simulatorKitHandle: UnsafeMutableRawPointer,
        via resolver: DlsymResolver
      ) throws
    }

    public struct DigitizerChannel: InputChannel { /* id=.digitizer, name="digitizer", requiredSymbolNames=DigitizerSymbolNames.allRequiredForC4 */ }
    public struct IndigoChannel:    InputChannel { /* id=.indigo9,   name="indigo9",   requiredSymbolNames=IndigoSymbolNames.allRequiredForC3 */ }
    ```

    **关键不变量**：
    1. `InputChannelId.rawValue` 字符串与 `SimxHostHIDArgs.Path.rawValue` 字面相同（`"digitizer"` / `"indigo9"`） — C5 `Path` 暂保留（不 retrofit C4 既有 args 形态），但**新代码引用统一走 `InputChannelId`**
    2. `DigitizerChannel.tap` 内部调 `DigitizerSymbols.resolve(...)` + `IOHIDDigitizerTap.tapNormalized(...)`（**不**改 `IOHIDDigitizerTap.swift` 一行）
    3. `IndigoChannel.tap` 内部调 `IndigoSymbols.resolve(...)` + `IndigoHIDDispatcher.tapNormalized(...)`（**不**改 `IndigoHIDDispatcher.swift` 一行）
    4. `requiredSymbolNames` 数组与 C3/C4 既有 `XxxSymbolNames.allRequiredForCN` 字面相同——`InputChannelTests` 锁定

- 文件 5（新建）：`swift-bridge/Sources/SimxIndigoHID/ChannelDispatcher.swift`
  - **锁形状**：
    ```swift
    public enum ChannelDispatcher {
      public struct Decision: Equatable {
        public let channel: InputChannelId
        public let warning: Warning?
      }
      public enum Warning: Equatable {
        case fiveArgIndigoNotImplemented(version: String)   // iOS 17/18 暂占位
      }
      public struct RuntimeVersion: Equatable {
        public let major: Int   // 17 / 18 / 26
        public let minor: Int   // 4 / 5 / ...
        public init(major: Int, minor: Int) { ... }
        public init?(identifier: String) { ... }            // 解析 "com.apple.CoreSimulator.SimRuntime.iOS-26-4" → (26, 4)
      }

      /// Version-dispatch decision table:
      ///   - explicit `override` → that channel, no warning
      ///   - major >= 26  → .digitizer (C4 path)
      ///   - major == 17 → .digitizer + warning .fiveArgIndigoNotImplemented (5-arg 推 v0.7)
      ///   - major == 18 → .digitizer + warning .fiveArgIndigoNotImplemented
      ///   - else (major < 17) → .digitizer + warning（最大努力）
      public static func decide(
        runtimeVersion: RuntimeVersion,
        override: InputChannelId?
      ) -> Decision

      /// Pure factory — no IO, no dlsym. Caller uses the returned channel
      /// to dispatch a tap (caller does dlopen + resolve + invoke).
      public static func channel(for id: InputChannelId) -> InputChannel
    }
    ```

    **关键不变量**：
    1. `decide(...)` 是**纯函数**（无 IO / 无 dlsym / 无 simctl spawn）—— 测试可零依赖注入
    2. 决策表与 plan 此章节 "Warning enum" 锁定形态字面一致 —— `ChannelDispatcherTests` 守护
    3. `Warning` 是 enum 不是 string —— 类型系统强制每个 case 配修复指引（CLAUDE.md §8.5）

    ```swift
    public struct ChannelProbeReport: Equatable {
      public struct ChannelEntry: Equatable {
        public let name: String          // == InputChannelId.rawValue
        public let available: Bool
        public let resolved: [String]    // 已解析的符号名
        public let missing:  [String]    // 缺失的符号名（按 requiredSymbolNames 顺序）
      }
      public let channels: [ChannelEntry]
      public var ok: Bool { channels.allSatisfy { $0.available } }

      /// Single-line wire JSON. Byte sequence is part of the contract with
      /// `src/cli/commands/doctor.ts` `JSON.parse` downstream.
      public func json() -> String

      /// Pure factory: builds a report by trying to resolve each channel's
      /// required symbols against the supplied handles. Catches per-channel
      /// dlsym failures and records them in `missing` — never throws.
      public static func probe(
        simulatorKitHandle: UnsafeMutableRawPointer,
        ioKitHandle: UnsafeMutableRawPointer,
        via resolver: DlsymResolver
      ) -> ChannelProbeReport
    }
    ```

  - **关键不变量**：
    1. `json()` 字节序列与 schema 锁定字面一致 — `ChannelProbeReportTests` 守护

## 关键技术决策（在 plan 里现场判定 — 决策影响 step 形态）

| # | 决策点 | 选 | 理由 |
|---|---|---|---|
| 1 | InputChannel 是 protocol vs class 还是 enum | protocol + 两个 struct 实现 | 多实现共存 + Swift idiom + 易 fake；enum 不 scale 到 5-arg channel；class 无 value semantics 收益 |
| 2 | `tap(...)` 签名要不要含 `simulatorKitHandle` / `coreSimulatorHandle` | 含 | dlopen 是 caller 职责（一处 dlopen 复用给 dispatch + probe，省一次 dlopen 开销 + 复用同 handle 避免 framework 多次加载状态不一致） |
| 3 | `probeAvailability` 返回值要不要含 error detail | 不含 | C5 doctor 上层只关心 boolean × 2；具体哪个 missing 写在 probe JSON `missing[]`；caller 端 detail 推 C6+ |
| 4 | `simx-host-hid tap` 缺省路径要不要改走 dispatcher decide | **不改**，仍硬编码 `.digitizer` | C4 既有 `simx-hostid-digitizer-smoke.sh` + 用户当前命令行习惯依赖 `.digitizer` 默认；C5 dispatcher 仅暴露 API + probe；CLI 默认行为 C6 接 Driver 时统一切换 |
| 5 | iOS 17/18 dispatcher 决策怎么处理（5-arg 未实现） | fallback `.digitizer` + return `Warning.fiveArgIndigoNotImplemented` | "fail with hint" 让 C7 e2e 在 17/18 sim 上无法跑——v0.2 出口验收要 17.5/18.4/26.x 全套通；"silent fallback"丢失诊断；带 Warning 的 fallback 是中间路线，doctor 不暴露 Warning（C5 doctor 只汇报 boolean），但 `tap()` 在 17/18 跑会 stderr 一行警告 |
| 6 | dispatcher 怎么拿到 iOS runtime version | 接口接受 `RuntimeVersion` 参数（caller 传） | 在 dispatcher 内 spawn simctl 会 (a) 单测难注入 (b) 增加 dispatcher 与 simctl IPC 耦合；C5 dispatcher 是纯函数，runtime 探测留给 caller（TS Driver 在 C6 接通时通过 SimSession.runtime 传入；C5 probe 不需要 runtime，所以暂无 caller） |
| 7 | `jsonEscape` 复用方式 | 提取 `internal enum SimxHostJsonEscape` | C3/C4 既有 `SimxHostHIDResult.jsonEscape` 是 private static —— probe 要复用必须 extract；新建 internal enum 比改 access modifier 更显式 |
| 8 | doctor `hid` check 失败时 exit code | 1（与既有 4 个失败 check 同形态） | doctor 是健康汇总，任一 check 失败 → exitCode=1；hid 不特殊 |
| 9 | doctor `hid` value 字段类型（dict vs string） | 改 `DoctorCheck.value` 从 `string` 到 `string \| HidChannelAvailability` | 唯一可行方案；保留向后兼容（既有 4 check value 仍 string）；TS exactOptionalPropertyTypes 下 union 显式好于 unknown |
| 10 | probe 二进制不存在的 doctor 行为 | check `hid` → `ok=false` + `message="hid probe failed: simx-host-hid not built at <path> (run: swift build --package-path swift-bridge)"` | 与 xcode probe spawn 失败 / claude 不存在 case 同形态；用户能 0-step 修复 |
