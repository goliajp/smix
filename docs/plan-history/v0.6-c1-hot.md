  import type { Server } from '@modelcontextprotocol/sdk/server/index.js'
  import { StdioServerTransport } from '@modelcontextprotocol/sdk/server/stdio.js'
  import type { Logger } from './server.js'

  export type McpCommandDeps = {
    server: Server
    transport?: unknown  // 测试注入；prod 走 StdioServerTransport 默认
    log?: Logger
    onSignal?: (cb: (sig: 'SIGINT' | 'SIGTERM') => void) => () => void
    stdin?: NodeJS.ReadableStream
  }

  export type CommandResult = { exitCode: number }

  export async function runMcpCommand(deps: McpCommandDeps): Promise<CommandResult> {
    const log = deps.log ?? ((msg) =>
      process.stderr.write(`[simx-mcp] ${msg}\n`))
    const transport = deps.transport ?? new StdioServerTransport()
    try {
      await deps.server.connect(transport as Parameters<Server['connect']>[0])
    } catch (err) {
      log(`server.connect failed: ${err instanceof Error ? err.message : String(err)}`)
      return { exitCode: 1 }
    }
    log('server connected via stdio')
    const stdin = deps.stdin ?? process.stdin
    await new Promise<void>((resolve) => {
      const subscribe = deps.onSignal ?? defaultSignalSubscribe
      const unsubscribe = subscribe((sig) => {
        log(`received ${sig}, shutting down`)
        resolve()
      })
      stdin.once('end', () => {
        log('stdin EOF, shutting down')
        unsubscribe()
        resolve()
      })
    })
    try {
      await deps.server.close()
    } catch (err) {
      log(`server.close error (ignored): ${err instanceof Error ? err.message : String(err)}`)
    }
    return { exitCode: 0 }
  }

  function defaultSignalSubscribe(
    cb: (sig: 'SIGINT' | 'SIGTERM') => void,
  ): () => void {
    const onSigInt = (): void => cb('SIGINT')
    const onSigTerm = (): void => cb('SIGTERM')
    process.on('SIGINT', onSigInt)
    process.on('SIGTERM', onSigTerm)
    return () => {
      process.off('SIGINT', onSigInt)
      process.off('SIGTERM', onSigTerm)
    }
  }

## v0.6 C1 smoke gate 字段（8 字段）

`scripts/simx-c1-mcp-smoke.sh` 输出单行 JSON：

1. `exit_code` (== 0 — 主子进程 graceful EOF close)
2. `initialize_ok` (== "ok" — initialize 响应 `serverInfo.name === "simx"`)
3. `tools_list_count` (== "ok" — `tools/list` 返 array length === 1)
4. `tools_list_has_ping` (== "ok" — `tools/list[0].name === "ping"`)
5. `ping_call_ok` (== "ok" — `tools/call ping` 返 `content[0].type === "text"`)
6. `ping_call_text` (== "ok" — `tools/call ping` 返 `content[0].text === "pong"`)
7. `malformed_handled` (== "ok" — 第二个子进程 malformed JSON 输入 → 返 JSON-RPC error code 数值 < 0)
8. `all_ok` (== "ok" — 上面 7 条全 ok 合取)

**PASS gate**：8 字段全 "ok" + exit 0。
