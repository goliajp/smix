  import type { Driver, LaunchOptions, FindOptions, SwipeDirection, KeyName, Permission, NetworkState, Appearance } from './types.js'
  import { ExpectationFailure } from '../core/index.js'
  import type { Selector, A11yNode, ScreenDescription } from '../core/index.js'
  import type { SimSession } from '../sim/session.js'
  import type { SimctlPermission } from '../sim/simctl.js'

  export class SimctlDriver implements Driver {
    private readonly session: SimSession
    private lastLaunchedBundleId: string | undefined

    constructor(session: SimSession) {
      this.session = session
      this.lastLaunchedBundleId = undefined
    }

    // 直通 client 的 11 个 A 组方法（签名匹配 Driver 接口；显式返回类型）
    async launch(bundleId: string, opts?: LaunchOptions): Promise<void> { ... }
    async terminate(bundleId: string): Promise<void> { ... }
    async install(path: string): Promise<void> { ... }
    async uninstall(bundleId: string): Promise<void> { ... }
    async openUrl(url: string): Promise<void> { ... }
    async pasteboardSet(text: string): Promise<void> { ... }
    async pasteboardGet(): Promise<string> { ... }
    async grantPermission(permission: Permission, bundleId?: string): Promise<void> { ... }
    async setAppearance(mode: Appearance): Promise<void> { ... }
    async screenshot(): Promise<Buffer> { ... }
    async describe(): Promise<ScreenDescription> { ... }

    // 18 个 B 组方法（实现体一行：return this.notSupported(...)）
    tap(_s: Selector): Promise<void>        { return this.notSupported('tap',        'HID bridge') }
    doubleTap(_s: Selector): Promise<void>  { return this.notSupported('doubleTap',  'HID bridge') }

      try { return await fn() } catch (err) {
        if (err instanceof ExpectationFailure) throw err  // 已是结构化的（如 grantPermission 缺 bundleId 自抛）
        const message = err instanceof Error ? err.message : String(err)
        throw new ExpectationFailure({
          code: 'DRIVER_ERROR',
          message: `Driver.${op}() failed: ${message}`,
          hint: `simctl call failed; check simulator state and bundle/path validity (op=${op})`,
        })
      }
    }
    ```

      const hintMap = {
        'HID bridge':         'requires HID bridge (v0.2); simctl does not expose touch/key injection',
        'AX bridge':          'requires AX bridge (v0.3); simctl does not expose element queries',
        'HID/system bridge':  'requires HID/system bridge (v0.2); simctl has no background/foreground subcommand',
        'not exposed by simctl': 'not exposed by simctl in v0.1; see v1.md section 2.A for the bridge roadmap',
      } as const
      return Promise.reject(new ExpectationFailure({
        code: 'DRIVER_ERROR',
        message: `Driver.${op}() not supported by SimctlDriver`,
        hint: hintMap[kind],
      }))
    }
    ```

### 5. SimctlClient 是否需要加 uninstall（C5 内做）
**是，加，在 S1 做**。理由：Driver 接口要求 `uninstall(bundleId)`，simctl 子命令 `simctl uninstall <udid> <bundleId>` 原生支持，wrapper 仅 1 行（`runText`），与 C3 同形态。**不**做"在 driver 内同样抛 DRIVER_ERROR"的替代方案——那是技术债，且与 install 不对称。S1 独立成步使 client 测试单文件可独立跑过、降耦合。

### 6. 错误转换策略（client 错误 → ExpectationFailure 怎么 wrap）
A 组所有 client 调用走 `private wrap<T>(op, fn)`：
- 捕到 `ExpectationFailure` 直接重抛（已结构化，如 grantPermission 自抛缺 bundleId）
- 捕到其他 Error → `new ExpectationFailure({ code: 'DRIVER_ERROR', message: \`Driver.${op}() failed: ${err.message}\`, hint: \`simctl call failed; check simulator state and bundle/path validity (op=${op})\` })`
- **不保留 cause 字段**（`FailureInit` 类型没有 cause；保留原 message 已足够 AI fix；若未来需 cause 用 `details` 扩展 — 不在 C5 范围）
- B 组所有方法走 `notSupported(op, kind)`：直接构造 `ExpectationFailure({ code: 'DRIVER_ERROR', message, hint })`，hint 按 4 个 kind 分桶（见 §"B. 未实现"表）。不与 client 交互（驱动测试断言 spawn/spawnBinary 未被调用）

### 7. describe() 的 frontApp 字段如何获取
**决策（已锁定）：在 driver 内部维护 `this.lastLaunchedBundleId`**：
- 构造时 `undefined`
- `launch(bundleId, ...)` 成功后写入 `bundleId`
- `terminate(bundleId)` **不**清空（多 app 场景下 terminate 仅意味"这个 app 关了"，不代表前台变成空；保持最后 launch 的 bundleId 作为 best-effort）
- `describe()` 返回 `frontApp: this.lastLaunchedBundleId ?? ''`（v0.1 未 launch 过返回空串，前端 / 上层可识别为"未知"，等 v0.3 AX 桥拿真 frontmost app 后替换）
- 不引入"driver 状态机"概念（避免过早抽象）；这就是一个 nullable 字段
- 替代方案 (b)"返回空串 + 留给 v0.3" 被拒：损失太多——C6 端到端 `screenshot-only` 需要在 launch 后能看到 bundleId 出现在 trace 里，验证 launch 真的发生
