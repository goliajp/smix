- **`.save` parser grammar**（决策 2：A discriminated union 内嵌 name 字段）：
  - `^\.save\s+as\s+(\S+)\s*$` regex 命中 → `{kind:'meta', verb:'.save', name: m[1]}`
  - `^\.save\s*$`（裸 `.save` 无 arg）→ `{kind:'error', message:".save requires 'as <name>' (e.g. .save as my-flow)"}`
  - `^\.save\s+(\S+)\s*$` 但缺 `as`（如 `.save foo`）→ 同上 hint
  - `^\.save\s+as\s+$`（`as` 后无 name）→ `{kind:'error', message:".save requires a name after 'as' (e.g. .save as my-flow)"}`
  - `^\.save\s+as\s+(\S+)\s+\S.*$`（多 trailing 字段，如 `.save as foo bar`）→ `{kind:'error', message:".save expects exactly one name argument"}`

  ```ts
  // 伪码示意——repl-codegen.ts 内
  export function generateTestTs(entries: HistoryEntry[], name: string): string {
    const lines: string[] = []
    lines.push("import { test } from '../src/sdk/index.js'")
    lines.push('')
    lines.push(`test('${escapeSingleQuote(name)}', async ({ app }) => {`)
    for (const e of entries) {
      const stmt = renderAction(e.parsed)
      if (stmt !== null) lines.push(`  ${stmt}`)
    }
    lines.push('})')
    lines.push('')  // trailing newline
    return lines.join('\n')
  }

  function renderAction(p: ParsedCommand): string | null {
    if (p.kind !== 'verb') return null  // defensive: callers should pass only verbs
    switch (p.verb) {
      case 'launch': return `await app.launch('${esc(p.arg)}')`
      case 'terminate': return `await app.terminate('${esc(p.arg)}')`
      case 'tap': return `await app.tap(${renderSelector(p.selector)})`
      case 'screenshot': return 'await app.screenshot()'
      case 'describe': return null  // 决策 4.e: describe 不入 TS
    }
  }

  function renderSelector(s: Selector): string {
    // literal mapping: parser 抓的字段直译；user 选择哪种 selector → TS 字面就是哪种
    const parts: string[] = []
    if (s.text !== undefined) parts.push(`text: '${esc(s.text)}'`)
    if (s.id !== undefined) parts.push(`id: '${esc(s.id)}'`)
    if (s.label !== undefined) parts.push(`label: '${esc(s.label)}'`)
    if (s.role !== undefined) parts.push(`role: '${esc(s.role)}'`)
    if (s.name !== undefined) parts.push(`name: '${esc(s.name)}'`)
    return `{ ${parts.join(', ')} }`
  }

  function esc(v: string): string {
    return v.replace(/\\/g, '\\\\').replace(/'/g, "\\'")
  }
  ```

- **`validateGeneratedTs(absPath)` 字面行为**（决策 5：A 自动同进程校验）：

  ```ts
  // 伪码示意
  export async function validateGeneratedTs(absPath: string): Promise<TscValidation> {
    return new Promise((resolve) => {
      const proc = spawn('bun', ['x', 'tsc', '-p', 'tsconfig.examples.json', '--noEmit'], {
        cwd: process.cwd(),
        stdio: ['ignore', 'pipe', 'pipe'],
      })
      let stdout = ''
      let stderr = ''
      proc.stdout.on('data', (b) => { stdout += b.toString() })
      proc.stderr.on('data', (b) => { stderr += b.toString() })
      proc.on('close', () => {
        const combined = stdout + stderr
        const hasErrors = /error TS\d+/.test(combined)
        if (hasErrors) resolve({ ok: false, output: combined })
        else resolve({ ok: true })
      })
      proc.on('error', (e) => {
        resolve({ ok: false, output: `tsc spawn failed: ${e.message}` })
      })
    })
  }
  ```

  - **同步 vs 异步**：异步 (`Promise<TscValidation>`)——tsc 单文件 ~1.2-2.5s，REPL 主循环 `pending.then` 已串行化 line event，await 等 tsc 完成不阻塞下一行（决策 5.a：A 阻塞 prompt 而非 fire-and-forget；理由：user 输入 `.save` 后等结果是合理 UX，与 `tap` 等 SDK 调用同形态；fire-and-forget 会让 user 在 `.exit` 后才看到 tsc 错误，feedback loop 断）
  - **`-p tsconfig.examples.json --noEmit`**：用 tsconfig.examples.json 才能覆盖 examples/ 路径（tsconfig.json `exclude: ["examples"]` 不覆盖）
  - **错误判定**：stdout/stderr 合并匹配 `/error TS\d+/`（tsc 标准错误格式 `path(line,col): error TS<code>: <msg>`）；`exitCode !== 0` 但无 `error TS` 视为环境问题（如 bun 不在 PATH）也走 fail path 同等处理
  - **stdio = ignore/pipe/pipe**：不让 tsc 子进程读 stdin（避免 readline 已占 stdin 的 conflict）+ 捕获 stdout/stderr 给 REPL 输出
  - **不污染 REPL prompt**：tsc 输出**不**直接流到 REPL output——而是收集后失败时按 head 10 行 indent 打到 REPL output；成功时仅打 `(tsc ok)` 一句

- **`isValidSaveName(name)` 白名单**（决策 3：A `^[a-z0-9][a-z0-9-]{0,63}$`）：

   ```ts
   import { spawn } from 'node:child_process'
   import type { ParsedCommand } from './repl-parser.js'
   import type { Selector } from '../../core/index.js'

   export type HistoryEntryLike = {
     seq: number
     raw: string
     parsed: ParsedCommand
   }

   export type TscValidation = { ok: true } | { ok: false; output: string }

   const NAME_RE = /^[a-z0-9][a-z0-9-]{0,63}$/

   export function isValidSaveName(name: string): boolean {
     return NAME_RE.test(name)
   }

   export function generateTestTs(
     entries: readonly HistoryEntryLike[],
     name: string,
   ): string {
     const lines: string[] = [
       "import { test } from '../src/sdk/index.js'",
       '',
       `test('${escapeSingleQuote(name)}', async ({ app }) => {`,
     ]
     for (const e of entries) {
       const stmt = renderAction(e.parsed)
       if (stmt !== null) lines.push(`  ${stmt}`)
     }
     lines.push('})')
     lines.push('')
     return lines.join('\n')
   }

   function renderAction(p: ParsedCommand): string | null {
     if (p.kind !== 'verb') return null
     switch (p.verb) {
       case 'launch':
         return `await app.launch('${escapeSingleQuote(p.arg)}')`
       case 'terminate':
         return `await app.terminate('${escapeSingleQuote(p.arg)}')`
       case 'tap':
         return `await app.tap(${renderSelector(p.selector)})`
       case 'screenshot':
         return 'await app.screenshot()'
       case 'describe':
         return null
     }
   }

   function renderSelector(s: Selector): string {
     const parts: string[] = []
     if (s.text !== undefined) parts.push(`text: '${escapeSingleQuote(String(s.text))}'`)
     if (s.id !== undefined) parts.push(`id: '${escapeSingleQuote(s.id)}'`)
     if (s.label !== undefined) parts.push(`label: '${escapeSingleQuote(s.label)}'`)
     if (s.role !== undefined) parts.push(`role: '${escapeSingleQuote(s.role)}'`)
     if (s.name !== undefined) parts.push(`name: '${escapeSingleQuote(String(s.name))}'`)
     return `{ ${parts.join(', ')} }`
   }

   function escapeSingleQuote(v: string): string {
     return v.replace(/\\/g, '\\\\').replace(/'/g, "\\'")
   }

   export async function validateGeneratedTs(
     absPath: string,
   ): Promise<TscValidation> {
     return new Promise((resolve) => {
       const proc = spawn('bun', ['x', 'tsc', '-p', 'tsconfig.examples.json', '--noEmit'], {
         cwd: process.cwd(),
         stdio: ['ignore', 'pipe', 'pipe'],
       })
       let stdout = ''
       let stderr = ''
       proc.stdout.on('data', (b: Buffer) => { stdout += b.toString() })
       proc.stderr.on('data', (b: Buffer) => { stderr += b.toString() })
       proc.on('close', () => {
         const combined = stdout + stderr
         if (/error TS\d+/.test(combined)) {
           resolve({ ok: false, output: combined })
         } else {
           resolve({ ok: true })
         }
       })
       proc.on('error', (e) => {
         resolve({ ok: false, output: `tsc spawn failed: ${e.message}` })
       })
     })
   }
   ```
