### S1. 创建 `swift-bridge/SimxRunner` xcodeproj + FlyingFox 依赖 + 永不结束的 test method（Swift XCTest 红→绿）

> **决策依据（见末尾"关键技术决策"表 1）**：用 `.xcodeproj` 而非 SPM Package。理由：XCUITest UI Testing bundle 在 Xcode 内是 first-class（target type `com.apple.product-type.bundle.ui-testing`），SPM 对 iOS UITest 支持仍不稳；`xcodebuild test -scheme ... -destination 'platform=iOS Simulator,id=...'` 是文档化路径。后续 v0.2 host-side Swift HID 代码（C3/C4）再用 SPM `swift-bridge/Package.swift`，二者并存（不冲突）。

**红（写 Swift 单元测试 — XCTest，host 侧跑，不上 simulator）**

> 路由 handler 是**纯函数**，host 侧 `swift build` 后 `swift test` 直接跑。**WHY**：把 handler 抽成 `HealthRoute.handle(_:) -> HTTPResponse`（输入 `HTTPRequest`，输出 `HTTPResponse`，不依赖 socket）→ 单测断言响应 body / status / content-type 全部不动 server。Server 启动这一段（`await server.run()`）放 simulator 内才能验，留 S2 端到端验。

- 新建 SPM Package（**这是 host-side library，与 S1 末尾的 xcodeproj 是两套东西**），位置 `swift-bridge/Package.swift`，含 1 个 library product `SimxRunnerCore`。`SimxRunnerCore` 提供 `HealthRoute`（路由 handler）+ 后续 v0.2 host-side bridge 代码的家。

  ```swift
  import FlyingFox
  import Foundation

  public enum HealthRoute {
    /// Pure function: produce the JSON body for GET /health.
    /// Kept pure so host-side XCTest can assert without spinning up a server.
    public static func body() -> Data {
      // Stable byte sequence — host-side curl will jq-parse it.
      return Data(#"{"ok":true}"#.utf8)
    }

    /// Build the HTTPResponse for GET /health.
    public static func response() -> HTTPResponse {
      return HTTPResponse(
        statusCode: .ok,
        headers: [.contentType: "application/json"],
        body: body()
      )
    }
  }
  ```

- 文件 3：`swift-bridge/SimxRunnerUITests/SimxRunnerUITests.swift`（新建）
  ```swift
  import XCTest
  import FlyingFox
  import SimxRunnerCore

  /// Long-running XCTestCase that hosts the in-simulator HTTP server.
  /// Pattern: Maestro re-building-the-ios-driver (https://maestro.dev/blog/maestro-re-building-the-ios-driver).
  /// The test method `test_runForever` intentionally never returns — `await server.run()`
  /// blocks until the host process kills the runner via `xcodebuild` cancellation.
  final class SimxRunnerUITests: XCTestCase {
    func test_runForever() async throws {
      let server = HTTPServer(port: 22087)
      await server.appendRoute("GET /health") { _ in
        HealthRoute.response()
      }
      try await server.run()
    }
  }
  ```

- **关键点**：
  - port `22087` 硬编码在 Swift 侧；TS 侧（C5+）需读同一常数 — C1 范围内只此一处，**不**抽常量
  - `0.0.0.0` 是 FlyingFox `HTTPServer(port:)` 默认监听 address（README 默认；C1 不显式指定 host）。**WHY**：simulator localhost = host localhost 在 iOS Simulator on macOS 上**默认共享**（同一 loopback namespace），host curl `127.0.0.1:22087` 可直达
  - test method 名 `test_runForever` 是为了让 `xcodebuild test` 命中（XCTest 发现规则：`test*`）；它不结束 → `xcodebuild` 命令将一直挂起（S2 用 `&` 后台 + 显式 PID 管理）
  - **不**写 `setUp` / `tearDown` — C1 不需要持久状态
  - **不**用 `XCUIApplication().launch()` —— C1 不交互 app，runner 跑起来即可（XCUITest 即便 host app 没活动也能起 server，因为 server live in *test runner* 进程）

- 文件 1：`scripts/simx-runner-health.sh`（新建 — repo 内 scripts/ 不存在则一并建）
  ```bash
  #!/usr/bin/env bash
  # simx-runner-health.sh — boot the SimxRunner XCUITest runner and curl /health.
  # WHY: v0.2 C1 verification harness; replaces hand-typing the long xcodebuild line.
  # Not wired into any TS/CLI yet (deferred to v0.2 C5+).
  set -euo pipefail

  ROOT="$(cd "$(dirname "$0")/.." && pwd)"
  PROJECT="$ROOT/swift-bridge/SimxRunner.xcodeproj"
  SCHEME="SimxRunner"
  PORT="${SIMX_RUNNER_PORT:-22087}"

  if [[ -z "${SIMX_UDID:-}" ]]; then
    SIMX_UDID="$(xcrun simctl list devices -j \
      | jq -r '[.devices | to_entries[] | .value[] | select(.isAvailable==true and .state=="Booted")][0].udid // empty')"
  fi
  if [[ -z "$SIMX_UDID" ]]; then
    echo "no booted iOS simulator (set SIMX_UDID or boot a sim)" >&2
    exit 2
  fi

  LOGDIR="$ROOT/.simx/runner"
  mkdir -p "$LOGDIR"
  LOGFILE="$LOGDIR/xcodebuild-$$.log"
  PIDFILE="$LOGDIR/runner-$$.pid"

  # Kick xcodebuild test in background; test_runForever never returns.
  ( xcodebuild -project "$PROJECT" \
               -scheme "$SCHEME" \
               -destination "platform=iOS Simulator,id=$SIMX_UDID" \
               -only-testing:SimxRunnerUITests/SimxRunnerUITests/test_runForever \
               test > "$LOGFILE" 2>&1 ) &
  echo $! > "$PIDFILE"

  cleanup() {
    local pid; pid="$(cat "$PIDFILE" 2>/dev/null || true)"
    if [[ -n "$pid" ]]; then kill -TERM "$pid" 2>/dev/null || true; fi
    # Also kill any descendant xcodebuild/simctl spawn that lingers.
    pkill -P "$pid" 2>/dev/null || true
  }
  trap cleanup EXIT

  # Poll /health up to 90s (cold install + boot may be slow).
  STATUS=""
  for i in $(seq 1 90); do
    if STATUS="$(curl -sS -o /tmp/simx-health-$$.body -w '%{http_code}' \
                       --max-time 2 "http://127.0.0.1:$PORT/health" 2>/dev/null)"; then
      if [[ "$STATUS" = "200" ]]; then
        BODY="$(cat /tmp/simx-health-$$.body)"
        echo "{\"status\":$STATUS,\"body\":$BODY}"
        rm -f /tmp/simx-health-$$.body
        exit 0
      fi
    fi
    sleep 1
  done

  echo "runner did not respond on :$PORT within 90s" >&2
  echo "--- xcodebuild log tail ---" >&2
  tail -40 "$LOGFILE" >&2 || true
  exit 1
  ```
