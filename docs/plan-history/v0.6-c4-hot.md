## 目标 checkpoint

**C4**：通过后世界变成 — `bun src/cli/index.ts mcp` 字面启动 stdio MCP server 同时 expose **19 个 tool**（C1 `ping` + C2 7 lifecycle + C3 4 observe + C4 新加 7 interaction），7 个新 tool 真接 `Driver` 接口：

- `tools/list` → 返 **19 element array**，名字字面：`ping` / `simulator_list` / `simulator_boot` / `simulator_shutdown` / `app_launch` / `app_terminate` / `app_install` / `app_uninstall` / `screen_describe` / `screen_screenshot` / `screen_hierarchy` / `element_inspect` / `tap` / `double_tap` / `long_press` / `fill` / `swipe` / `scroll_to` / `key_press`
- `tools/call name=tap arguments={udid:"...", selector:{text:"Settings"}}` → 调 `driver.tap({text:"Settings"})` → 再调 `driver.describe()` → 返 `{content:[{type:'text', text:'{"ok":true, "screen":{...ScreenDescription}}'}]}`
- `tools/call name=tap arguments={udid:"...", selector:{text:"Settings"}, returnSummary:false}` → 调 `driver.tap(...)` 后**跳过** describe → 返 `{content:[{type:'text', text:'{"ok":true, "screen":null}'}]}`（决策 7.A 字面）
- `tools/call name=double_tap arguments={udid:"...", selector:{id:"map"}}` → 调 `driver.doubleTap({id:"map"})` + describe → `{ok:true, screen}`
- `tools/call name=long_press arguments={udid:"...", selector:{text:"Item"}, durationMs:1500}` → 调 `driver.longPress({text:"Item"}, 1500)` + describe → `{ok:true, screen}`
- `tools/call name=fill arguments={udid:"...", selector:{id:"email"}, value:"foo@bar.com"}` → 调 `driver.fill({id:"email"}, "foo@bar.com")` + describe → `{ok:true, screen}`（决策 5.A 字面 — MCP wire `value` → driver `text`）
- `tools/call name=swipe arguments={udid:"...", direction:"up"}` → 调 `driver.swipe("up")` + describe → `{ok:true, screen}`
- `tools/call name=swipe arguments={udid:"...", direction:"left", from:{text:"Card"}}` → 调 `driver.swipe("left", {text:"Card"})` + describe → `{ok:true, screen}`
- `tools/call name=swipe arguments={udid:"...", direction:"diagonal"}` → 不符合 `enum(['up','down','left','right'])` zod refine → `{isError:true, content:[{type:'text', text:'invalid arguments: ...'}]}`（决策 3.A 字面）
- `tools/call name=scroll_to arguments={udid:"...", selector:{text:"Bottom"}}` → 调 `driver.scrollTo({text:"Bottom"})` + describe → `{ok:true, screen}`
- `tools/call name=key_press arguments={udid:"...", key:"return"}` → 调 `driver.pressKey("return")` + describe → `{ok:true, screen}`
- `tools/call name=key_press arguments={udid:"...", key:"enter"}` → 不符合 `enum(['return','delete','tab','space','escape','arrowUp','arrowDown','arrowLeft','arrowRight'])` zod refine → `{isError:true, content:[{type:'text', text:'invalid arguments: ...'}]}`（决策 4.A 字面 — `enter` 不在 9 字面中；`return` 才是字面值）
- 任一 interaction tool 主操作失败（driver acquire throw / driver.tap throw / 选 selector 无效 zod 校验失败）→ `{isError:true, content:[{type:'text', text:'<error message>'}]}`（C2 决策 10.A / C3 决策延续）

    // Run a driver mutation, then optionally re-describe to give AI the new screen
    // in a single round-trip. If the mutation succeeds but describe throws, the
    // tool still reports ok:true with screen:null + screenError — the user-visible
    // action already happened; surfacing it as isError would hide that.
    async function withScreenAfter(
      ctx: ToolContext,
      udid: string,
      returnSummary: boolean,
      mutate: (driver: Driver) => Promise<void>,
    ): Promise<ToolCallResult> {
      let driver: Driver
      try {
        driver = await ctx.acquireDriver(udid)
      } catch (err) {
        return textError(errMsg(err))
      }
      try {
        await mutate(driver)
      } catch (err) {
        return textError(errMsg(err))
      }
      if (!returnSummary) {
        return textOk({ ok: true, screen: null })
      }
      try {
        const screen = await driver.describe()
        return textOk({ ok: true, screen })
      } catch (err) {
        return textOk({ ok: true, screen: null, screenError: errMsg(err) })
      }
    }
