> **决策依据（见末尾"关键技术决策" 表 #1-#5）**：
> - **wire JSON schema 与 ChannelProbeReport 形似但不同 type**（决策 #1）：HID probe 是"C 符号 dlsym 在 / 不在"；AXP probe 是"framework loaded + ObjC class load + +sharedInstance IMP + instance selector responds"。**两者 schema 共享 `{ok: bool}` 顶层 + per-category `{resolved:[], missing:[]}` 子结构**，但顶层 keys 不同：HID `channels: [{name,available,resolved,missing}]`；AXP `framework: {path,loaded}` + `classes: {resolved,missing}` + `selectors: {resolved,missing}` + `sharedInstance: {available}`。**不**复用 ChannelEntry struct（拒绝 forced abstraction —— 一年后回看两 probe 的 schema 性质不同，硬合并是负债；YAGNI）。
> - **probe 用同样的 `SystemDlsymResolver`**（决策 #2）：复用 v0.2 C3 `DlsymResolver` protocol；framework `dlopen` 走绝对路径 `/System/Library/PrivateFrameworks/AccessibilityPlatformTranslation.framework/AccessibilityPlatformTranslation`（baguette 同路径，已实测）；`RTLD_NOW | RTLD_GLOBAL`（baguette 使用，确保后续 NSClassFromString 能见到符号）。
> - **class load 用 `NSClassFromString`**（决策 #3）：与 v0.2 C3 `CoreSimulatorBridge.resolveSimDevice` 同模式（CLAUDE.md §9.6 "动态加载，不硬链接"）。**不** `import AccessibilityPlatformTranslation`（私有 framework，无 modulemap；硬 import 会 build fail）。
> - **selector responds 用 `class_getMethodImplementation` + `(instance as NSObject).responds(to:)`**（决策 #4）：与 baguette 同模式；`+sharedInstance` 必须能调出非 nil 实例才能 `responds(to:)`（dead class 无法 respond）；C2 invoke `+sharedInstance` 是**合法**的（baguette 注释 "Critical: install the token delegate" 说明真正会触发 XPC 路由失败的是后续 setValue bridgeTokenDelegate + invoke instance method，**单纯 +sharedInstance 不触发 XPC**）。
> - **probe 失败时报告全 false**（决策 #5）：与 `ChannelProbeReport.allUnavailable` 同模式；JSON wire 仍 well-formed；exit 0（probe 永不失败，能跑出 JSON 就是成功）。
> - **AppKit linked 进 SimxHostHID executable**（决策 #6）：当前 `SimxHostHID` executableTarget 已 link `SimxIndigoHID`（透传 Foundation + ObjectiveC.runtime）；AXP framework 在 dlopen 后 NSClassFromString 即可 load classes ——**无需** import AppKit / 改 Package.swift dependencies。v0.2 C5 已实测 framework class 在 hostHID 进程内通过 NSClassFromString 拿。
> - **CGRect / 坐标转换**（决策 #7）：C2 **不**做坐标转换（仅 probe）；v0.7+ 真读 tree 时再面对 host pt ↔ device pt 问题（baguette `AXFrameTransform`）。

    ```swift
    public static func probeSharedInstance(on cls: AnyClass?) -> SharedInstanceProbe {
      guard let cls = cls else { return .init(available: false) }
      let sel = NSSelectorFromString("sharedInstance")
      guard let metaCls = object_getClass(cls),
            let imp = class_getMethodImplementation(metaCls, sel) else {
        return .init(available: false)
      }
      typealias Fn = @convention(c) (AnyClass, Selector) -> AnyObject?
      let inst = unsafeBitCast(imp, to: Fn.self)(cls, sel)
      return .init(available: inst != nil)
    }
    ```

    ```swift
    public static func probeInstanceSelectors(names: [String], on instance: AnyObject?) -> SelectorsProbe {
      guard let instance = instance else {
        return .init(resolved: [], missing: names)
      }
      var resolved: [String] = []
      var missing: [String] = []
      for n in names {
        let sel = NSSelectorFromString(n)
        if (instance as AnyObject).responds(to: sel) {
          resolved.append(n)
        } else {
          missing.append(n)
        }
      }
      return .init(resolved: resolved, missing: missing)
    }
    ```

    ```swift
    public static func probeLive(via resolver: DlsymResolver) -> AxpCapabilityReport {
      // RTLD_GLOBAL required so NSClassFromString can resolve AXP symbols
      // after dlopen — see decision #8 in plan-hot.
      let handle = frameworkPath.withCString { Darwin.dlopen($0, RTLD_NOW | RTLD_GLOBAL) }
      let framework = FrameworkProbe(path: frameworkPath, loaded: handle != nil)
      guard handle != nil else { return AxpCapabilityReport.allUnavailable() }

      let classes = probeClasses(names: requiredClassNames, using: NSClassFromString)
      let translatorCls = NSClassFromString("AXPTranslator")
      let sharedInst = probeSharedInstance(on: translatorCls)
      // Invoke +sharedInstance once to get an instance for selector probing.
      let instance: AnyObject? = {
        guard sharedInst.available, let cls = translatorCls else { return nil }
        let sel = NSSelectorFromString("sharedInstance")
        guard let metaCls = object_getClass(cls),
              let imp = class_getMethodImplementation(metaCls, sel) else { return nil }
        typealias Fn = @convention(c) (AnyClass, Selector) -> AnyObject?
        return unsafeBitCast(imp, to: Fn.self)(cls, sel)
      }()
      let selectors = probeInstanceSelectors(names: requiredInstanceSelectorNames, on: instance)

      return AxpCapabilityReport(
        framework: framework, classes: classes, selectors: selectors, sharedInstance: sharedInst
      )
    }
    ```

## `AxpCapabilityProbe.probeLive` 伪代码（Swift 侧）

> 位置：`swift-bridge/Sources/SimxIndigoHID/AxpCapabilityProbe.swift`，public static method。

```swift
// In AxpCapabilityProbe.swift — production entrypoint.
// Composes 4 sub-probes against the live macOS runtime. Pure side-effect
// is the dlopen — every subsequent step is read-only.
// RTLD_GLOBAL is REQUIRED: without it, NSClassFromString fails to find
// AXP classes even after dlopen succeeds (baguette comment confirms).
// CLAUDE.md §9.6 invariant maintained: no `import AccessibilityPlatformTranslation`,
// no `linkedFramework`, all access via dlopen + NSClassFromString +
// class_getMethodImplementation. Symmetric to v0.2 C3 CoreSimulatorBridge.

public static func probeLive(via resolver: DlsymResolver) -> AxpCapabilityReport {
  // ---- framework ----
  let handle = frameworkPath.withCString { Darwin.dlopen($0, RTLD_NOW | RTLD_GLOBAL) }
  let framework = FrameworkProbe(path: frameworkPath, loaded: handle != nil)
  guard handle != nil else { return .allUnavailable() }

  // ---- classes ----
  let classes = probeClasses(names: requiredClassNames, using: NSClassFromString)

  // ---- +sharedInstance ----
  let translatorCls = NSClassFromString("AXPTranslator")
  let shared = probeSharedInstance(on: translatorCls)

  // ---- instance selector responds ----
  // Only invoke +sharedInstance for selector probing — does NOT trigger
  // XPC routing (XPC requires bridgeTokenDelegate + token registration,
  // which we do not install; that's v0.7+ host-side tree read scope).
  let instance: AnyObject? = {
    guard shared.available, let cls = translatorCls else { return nil }
    let sel = NSSelectorFromString("sharedInstance")
    guard let metaCls = object_getClass(cls),
          let imp = class_getMethodImplementation(metaCls, sel) else { return nil }
    typealias Fn = @convention(c) (AnyClass, Selector) -> AnyObject?
    return unsafeBitCast(imp, to: Fn.self)(cls, sel)
  }()
  let selectors = probeInstanceSelectors(
    names: requiredInstanceSelectorNames, on: instance
  )

  return AxpCapabilityReport(
    framework: framework, classes: classes,
    selectors: selectors, sharedInstance: shared
  )
}
```
