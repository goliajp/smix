  - **锁形状（plan 不写实现，代码作者自写函数体；以下是 API + 不变量）**：
    ```swift
    import Foundation
    import CoreGraphics

    /// IOHIDEvent-tree digitizer tap. Avoids the iOS 26.4 mouseFn-9-arg
    /// regression (events get rerouted to Home gesture or dropped). Mirrors
    /// baguette IOHIDDigitizerDispatch architecture: build parent digitizer
    /// + finger child IOHIDEvent → IndigoHIDMessageForTrackpadEventFromHIDEventRef
    /// → byte-patch (target tag 0x32 + edge bitmask) → send via the same
    /// SimDeviceLegacyHIDClient.sendWithMessage:... selector C3 uses.
    public enum IOHIDDigitizerTap {

      public enum Edge: Equatable {
        case none, left, top, right, bottom
        public var bit: UInt8 { /* none=0x00 left=0x02 top=0x08 right=0x04 bottom=0x01 */ }
      }

      public enum Phase: Equatable {
        case down, move, up
        public var eventMask: UInt32 { /* down/move=0x07, up=0x06 */ }
        public var range: Bool { /* !up */ }
        public var touch: Bool { /* !up */ }
      }

      /// Public tap entry — analogous to `IndigoHIDDispatcher.tapNormalized`.
      public static func tapNormalized(
        udid: String,
        x: Double, y: Double,
        symbols: IndigoSymbols,                 // reuse C3 IndigoSymbols (only needs nothing from it actually — but keeps API symmetric for InputChannel C5)
        digitizerSymbols: DigitizerSymbols,
        developerDir: String,
        coreSimulatorHandle: UnsafeMutableRawPointer
      ) throws

      // Pure function (testable): patch one IOHIDEvent message blob in-place.
      // `size` injected for unit-test reproducibility (production calls
      // `malloc_size(msg)`; tests pass an allocated buffer size).
      internal static func patch(
        message: UnsafeMutableRawPointer,
        edge: Edge,
        size: Int
      )
    }

    /// 4 C symbols required for the digitizer path. 3 live in IOKit
    /// (public framework, resolvable via dlsym(RTLD_DEFAULT) — they're in
    /// the dyld shared cache); 1 lives in SimulatorKit (private, requires
    /// explicit dlopen on `<dev>/Library/PrivateFrameworks/SimulatorKit.framework/SimulatorKit`).
    public struct DigitizerSymbols {
      public let createDigitizerEvent: UnsafeMutableRawPointer
      public let createFingerEvent:    UnsafeMutableRawPointer
      public let appendEvent:          UnsafeMutableRawPointer
      public let trackpadWrap:         UnsafeMutableRawPointer

      public static func resolve(
        ioKitHandle: UnsafeMutableRawPointer,   // RTLD_DEFAULT sentinel (-2)
        simulatorKitHandle: UnsafeMutableRawPointer,
        via resolver: DlsymResolver
      ) throws -> DigitizerSymbols
    }

    public struct DigitizerSymbolNames {
      public static let createDigitizerEvent =
        "IOHIDEventCreateDigitizerEvent"
      public static let createFingerEvent    =
        "IOHIDEventCreateDigitizerFingerEvent"
      public static let appendEvent          = "IOHIDEventAppendEvent"
      public static let trackpadWrap         =
        "IndigoHIDMessageForTrackpadEventFromHIDEventRef"
      public static let allRequiredForC4: [String] = [
        createDigitizerEvent, createFingerEvent, appendEvent, trackpadWrap,
      ]
    }

    /// Byte-level ABI constants — pinned by `IOHIDDigitizerTapTests`.
    public enum DigitizerConstants {
      public static let indigoHIDTouchTarget:        UInt32 = 0x32
      public static let transducerFinger:            UInt32 = 2
      public static let kTargetByteOffset0:          Int    = 0x6c
      public static let kTargetByteOffset1:          Int    = 0x10c
      public static let kEdgePresentByteOffsetA:     Int    = 0x3a
      public static let kEdgeBitByteOffsetA:         Int    = 0x3b
      public static let kEdgePresentByteOffsetB:     Int    = 0xda
      public static let kEdgeBitByteOffsetB:         Int    = 0xdb
      public static let kMinSizeForSecondTargetSlot: Int    = 0x110
      public static let kMinSizeForSecondEdgeSlot:   Int    = 0xdc
    }

    /// Typealiases — exact arg shapes from baguette / IOKit headers.
    public typealias CreateDigitizerEventFn = @convention(c) (
      CFAllocator?, UInt64,
      UInt32, UInt32, UInt32, UInt32, UInt32,
      Double, Double, Double, Double, Double,
      Bool, Bool, UInt32
    ) -> Unmanaged<CFTypeRef>?

    public typealias CreateFingerEventFn = @convention(c) (
      CFAllocator?, UInt64,
      UInt32, UInt32, UInt32,
      Double, Double, Double, Double, Double,
      Bool, Bool, UInt32
    ) -> Unmanaged<CFTypeRef>?

    public typealias AppendEventFn   = @convention(c) (CFTypeRef, CFTypeRef, UInt32) -> Void
    public typealias TrackpadWrapFn  = @convention(c) (UnsafeRawPointer) -> UnsafeMutableRawPointer?
    ```

  - **关键不变量**：
    1. byte-offset 常量与 baguette 完全一致（0x6c / 0x10c / 0x3a / 0x3b / 0xda / 0xdb）
    2. target tag `0x32` 与 C3 `IndigoConstants.touchDigitizer` 字面相同（路由到同一 digitizer subsystem，wire ABI 复用）
    3. **dlsym IOKit 用 `RTLD_DEFAULT` 哨兵**（`UnsafeMutableRawPointer(bitPattern: -2)`）— baguette 实测；IOKit 是公开 framework + 已在 dyld shared cache，无需 `dlopen`；**但 CLAUDE.md §9.6 私有符号 dlsym 不变量**对**公开**符号同样合规（依然 dlsym 动态获取，不 import IOKit 模块、不 linkedFramework 写进 Package.swift）
    4. SimulatorKit trackpad wrap 仍走 `dlopen` 绝对路径（与 C3 SimulatorKit 加载一致）
    5. `sendMessage` 复用 C3 `IndigoHIDDispatcher` 中的 selector helper — S1 必须**提取**该 helper 到 internal type（见下方"重构"段）

## 与冷计划假设不符之处（必须列出，不要隐瞒）

> 冷计划 v0.2 原 checkpoint 概要：C1 runner / C2 XCUITest tap fallback / C3 = 5-arg Indigo / C4 = 9-arg Indigo / C5 = InputChannel + version dispatch / C6 = TS Driver 接通 / C7 = e2e smoke。C3 已调换为 9-arg，C4 现在是**第二次偏离**——digitizer 路径不在冷计划任一 checkpoint 原文中。

1. **冷计划 C4 概要"9-arg Indigo HID 跑通"**：C3 已实现 9-arg 私有 framework infra（dlopen + dlsym + ObjC selector + IndigoHIDMessageForMouseNSEvent 9-arg shape），但 mouseFn 直路径在 iOS 26.4 不真触发 UI（v1.md 2026-05-15 决策日志已记录）。**C4 改 port baguette IOHIDDigitizerDispatch** 作为 iOS 26+ 主路径——这超出原冷计划字面，但**仍在 v1.md §2.A 表格"iOS 26+ 9-arg Indigo, digitizer target 0x32"边界内**（digitizer target = 0x32 与 IOHIDEvent 树路径里的 byte 0x6c/0x10c 写入 0x32 字面一致；wire ABI 同一 target）。决策日志记录扩展。
